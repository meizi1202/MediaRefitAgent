```python
def execute_xxx(state: VideoAgentState) -> VideoAgentState:
    """执行你的功能"""
    video_path = state.get("temp_video_path") or state.get("video_path")
    if not video_path:
        state["error"] = "视频文件不存在"
        state["current_step"] = "confirm_complete"
        return state
    try:
        # 你的处理逻辑
        output_path = "..."
        state["current_step"] = "confirm_complete"
        _append_message(state, "assistant", f"处理完成！\n[PREVIEW:{output_path}]")
    except Exception as e:
        state["error"] = str(e)
        state["current_step"] = "confirm_complete"
        _append_message(state, "assistant", f"处理异常: {str(e)}")
    return state
```
