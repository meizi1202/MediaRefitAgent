## Context

用户选择**前后端分离 + Vue.js**架构：
- 前端：Vue.js 3 单页应用（SPA）
- 后端：保持现有 FastAPI不变，仅作为 API 服务

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { Session, Message, Feature, VideoResult } from '../types';

export const useAppStore = defineStore('app', () => {
  // 状态
  const sessions = ref<Session[]>([]);
  const currentSessionId = ref<string | null>(null);
  const currentFeature = ref<Feature>('orient');
  const currentVideoData = ref<VideoResult | null>(null);

  // 计算属性
  const currentSession = computed(() =>
    sessions.value.find(s => s.session_id === currentSessionId.value)
  );

  const currentMessages = computed(() =>
    currentSession.value?.messages || []
  );

  // Actions
  function addMessage(sessionId: string, message: Message) {
    const session = sessions.value.find(s => s.session_id === sessionId);
    if (session) {
      session.messages.push(message);
    }
  }

  function setCurrentSession(sessionId: string | null) {
    currentSessionId.value = sessionId;
  }

  function setFeature(feature: Feature) {
    currentFeature.value = feature;
  }

  function setVideoData(data: VideoResult | null) {
    currentVideoData.value = data;
  }

  return {
    sessions,
    currentSessionId,
    currentFeature,
    currentVideoData,
    currentSession,
    currentMessages,
    addMessage,
    setCurrentSession,
    setFeature,
    setVideoData,
  };
});

## Phase 5: Composables

**新增** `frontend/src/composables/useSessions.ts`：
```typescript
import { ref } from 'vue';
import { api } from '../api';
import type { Session, Message } from '../types';

export function useSessions() {
  const sessions = ref<Session[]>([]);
  const currentSessionId = ref<string | null>(null);

  async function loadSessions() {
    try {
      const data = await api.getSessions();
      sessions.value = data.sessions || [];
    } catch (e) {
      console.error('Failed to load sessions:', e);
    }
  }

  async function createSession() {
    currentSessionId.value = null;
    sessions.value.unshift({
      session_id: Date.now().toString(),
      name: '新会话',
      messages: [],
      created: new Date().toISOString(),
    });
  }

  async function selectSession(sessionId: string) {
    currentSessionId.value = sessionId;
    const session = sessions.value.find(s => s.session_id === sessionId);
    if (session && session.messages.length === 0) {
      try {
        const data = await api.getSession(sessionId);
        session.messages = data.messages || [];
      } catch (e) {
        console.error('Failed to load session:', e);
      }
    }
  }

  async function deleteSession(sessionId: string) {
    try {
      await api.deleteSession(sessionId);
      sessions.value = sessions.value.filter(s => s.session_id !== sessionId);
      if (currentSessionId.value === sessionId) {
        currentSessionId.value = null;
      }
    } catch (e) {
      console.error('Failed to delete session:', e);
    }
  }

  function addMessage(sessionId: string, message: Message) {
    const session = sessions.value.find(s => s.session_id === sessionId);
    if (session) {
      session.messages.push(message);
    }
  }

  return {
    sessions,
    currentSessionId,
    loadSessions,
    createSession,
    selectSession,
    deleteSession,
    addMessage,
  };
}
```

export function useVideo() {
  const selectedFile = ref<File | null>(null);
  const selectedFiles = ref<File[]>([]);
  const videoData = ref<VideoResult | null>(null);
  const isLoading = ref(false);

  function selectFile(file: File | null) {
    selectedFile.value = file;
  }

  function selectFiles(files: File[]) {
    selectedFiles.value = files;

    isLoading.value = true;

      const result = await api.transform(formData);
      videoData.value = result;
      return result;
    } finally {
      isLoading.value = false;
    }
  }

  async function compress(formData: FormData) {
    isLoading.value = true;
    try {
      const result = await api.compress(formData);
      videoData.value = result;
      return result;
    } finally {
      isLoading.value = false;
    }
  }

      const result = await api.trim(formData);

      return result;
    } finally {
      isLoading.value = false;
    }
  }

    isLoading.value = true;
    try {
      const result = await api.concat(formData);
      videoData.value = result;
      return result;
    } finally {
      isLoading.value = false;

    videoData.value = null;

# MediaRefitAgent Agent 模块优化

## 背景

Agent 目录原为单文件（1339 行 video_agent.py + 666 行 langchain_agent.py），职责混杂不利于维护。

## 优化后结构

```
agent/
├── __init__.py          # 统一导出
├── types.py             # 共享类型定义（VideoAgentState, ConversationMessage）
├── video_agent.py       # 状态机编排入口（378 行）
├── langchain_agent.py   # 意图解析逻辑（570 行）
├── prompts.py           # 提示词模板
├── cli.py               # 命令行接口
├── llm_client.py        # LLM 客户端（向后兼容）
├── interfaces.py        # Protocol 接口（依赖注入）
├── nodes/               # Node 函数拆分
│   ├── __init__.py
│   ├── analyze.py       # 意图分析 Node
│   ├── detect.py        # 视频检测 Node
│   ├── execute.py       # 执行类 Node（transform/compress/trim/concat）
│   └── routing.py       # 路由和用户响应处理
├── memory/              # 会话历史存储
│   ├── __init__.py
│   └── store.py
└── llm/                 # LLM Provider 分离
    ├── __init__.py
    └── minimax.py       # MinMax LLM 实现
```

## 拆分说明

### 1. nodes/ - Node 函数拆分

| 文件 | 职责 | 包含函数 |
|------|------|----------|
| `analyze.py` | 意图分析 | `analyze_intent`, `IntentParser`, `_parse_ui_params` |
| `detect.py` | 视频检测 | `detect_video` |
| `execute.py` | 执行转换 | `execute_transform`, `execute_compress`, `execute_trim`, `execute_concat`, `confirm_complete` |
| `routing.py` | 路由逻辑 | `handle_user_response`, `should_proceed` |

### 2. memory/ - 会话历史存储

将 `SessionHistoryStore` 和 `MinMaxChatHistory` 从 `langchain_agent.py` 分离，便于：
- 支持多种存储后端（内存/Redis/数据库）
- 独立测试和维护

### 3. llm/ - LLM Provider 分离

```
llm/
├── __init__.py
└── minimax.py    # MinMax LLM 实现
```

统一 LLM 接口，便于切换 provider（如 OpenAI、Claude 等）。

### 4. types.py - 共享类型

避免循环导入，将 `VideoAgentState` 和 `ConversationMessage` 独立存放。

## 关键设计

### 循环导入解决

```
video_agent.py → nodes/__init__.py → nodes/analyze.py
                                            ↓
                                    agent.types (共享类型)
```

- `types.py` 定义共享类型，不依赖其他模块
- Node 函数从 `agent.types` 导入类型
- `video_agent.py` 延迟导入 nodes，避免启动时循环
